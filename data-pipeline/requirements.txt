# Auralis — data-pipeline-only requirements
# These are NOT installed on the production server. The pipeline runs on the
# developer workstation; its outputs (data/manifest.json + per-clip embedding
# JSONs) are committed to the repo and served as static assets in production.

numpy==1.26.4
scipy==1.14.1
scikit-learn==1.5.2
librosa==0.10.2.post1
soundfile==0.12.1
matplotlib==3.9.2

# Optional: faster non-linear projection (UMAP). Falls back gracefully when
# missing; the pipeline always provides PCA and t-SNE via scikit-learn.
umap-learn==0.5.6

# Optional: deep pitch tracker. Heavy install (~134 MB model + tensorflow
# ~350 MB), but dramatically more accurate than `librosa.piptrack` on
# voiced / harmonic content. Fallback is automatic when missing.
#
# CREPE's setup.py uses `pkg_resources`, so install with setuptools<81
# and pip's --no-build-isolation:
#   pip install 'setuptools<81'
#   pip install --no-build-isolation --no-deps crepe
#   pip install hmmlearn imageio resampy tensorflow
crepe==0.0.16
hmmlearn==0.3.3
tensorflow==2.21.0

# Optional: YAMNet (Hershey et al., 2017) pretrained deep audio embeddings
# via TensorFlow Hub. Adds a "yamnet" track in the manifest, projected to
# 6D via corpus-wide PCA. Requires tensorflow (already pinned above).
tensorflow-hub==0.16.1

# Optional: CLAP (Wu et al. 2023) language-audio embeddings via Hugging
# Face transformers. Adds a "clap" track in the manifest, projected to
# 6D via corpus-wide PCA. Model `laion/clap-htsat-unfused` is ~600 MB
# on first download; torch is ~700 MB. Pipeline-only — these libraries
# don't ship to production. Set AURALIS_SKIP_CLAP=1 to opt out during
# a quick rebuild.
#
# Install with:
#   pip install torch transformers
# (CPU-only is fine; CUDA is faster but not required at this corpus size.)
torch>=2.2
transformers>=4.40

# Optional: bundled ffmpeg binary used by `curated_downloads.py` to
# transcode OGG-wrapped FLAC files (e.g. some NASA clips on Wikimedia)
# into OGG/Vorbis so `libsndfile` can decode them at ingest time.
# Without this, those clips download but get skipped at ingest with a
# "flac decoder lost sync" error. Tiny dep (~50 MB binary).
imageio-ffmpeg>=0.5
